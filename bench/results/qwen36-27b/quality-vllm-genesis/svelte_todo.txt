```svelte
<script lang="ts">
  import { onMount } from 'svelte';

  type Task = {
    id: number;
    text: string;
    completed: boolean;
  };

  let tasks: Task[] = $state([]);
  let newTask = $state('');
  let filter = $state<'all' | 'active' | 'completed'>('all');

  function loadTasks(): void {
    try {
      const stored = localStorage.getItem('svelte-tasks');
      if (stored) {
        tasks = JSON.parse(stored);
      }
    } catch {
      tasks = [];
    }
  }

  function saveTasks(): void {
    localStorage.setItem('svelte-tasks', JSON.stringify(tasks));
  }

  onMount(() => {
    loadTasks();
  });

  function addTask(): void {
    const trimmed = newTask.trim();
    if (!trimmed) return;
    tasks.push({ id: Date.now(), text: trimmed, completed: false });
    newTask = '';
    saveTasks();
  }

  function toggleTask(task: Task): void {
    task.completed = !task.completed;
    saveTasks();
  }

  function deleteTask(id: number): void {
    tasks = tasks.filter((t) => t.id !== id);
    saveTasks();
  }

  function clearCompleted(): void {
    tasks = tasks.filter((t) => !t.completed);
    saveTasks();
  }

  $derived(
    filteredTasks =
      filter === 'active'
        ? tasks.filter((t) => !t.completed)
        : filter === 'completed'
        ? tasks.filter((t) => t.completed)
        : tasks
  );

  $derived(remaining = tasks.filter((t) => !t.completed).length);
</script>

<div class="min-h-screen bg-gradient-to-br from-slate-100 to-slate-200 flex items-start justify-center p-4 sm:p-8">
  <div class="w-full max-w-lg">
    <h1 class="text-4xl font-bold text-center text-slate-700 mb-8 tracking-tight">Todos</h1>

    <div class="bg-white rounded-xl shadow-lg border border-slate-200 overflow-hidden">
      <!-- Input -->
      <div class="p-4 border-b border-slate-100">
        <form
          onsubmit={(e) => {
            e.preventDefault();
            addTask();
          }}
          class="flex gap-2"
        >
          <input
            type="text"
            bind:value={newTask}
            placeholder="What needs to be done?"
            class="flex-1 px-4 py-3 rounded-lg border border-slate-300 focus:outline-none focus:ring-2 focus:ring-indigo-400 focus:border-transparent text-slate-700 placeholder-slate-400"
          />
          <button
            type="submit"
            class="px-5 py-3 bg-indigo-500 hover:bg-indigo-600 text-white font-medium rounded-lg transition-colors duration-200"
          >
            Add
          </button>
        </form>
      </div>

      <!-- Task List -->
      <ul class="divide-y divide-slate-100">
        {#each filteredTasks as task (task.id)}
          <li
            class="flex items-center gap-3 px-4 py-3 hover:bg-slate-50 transition-colors duration-150 group"
          >
            <input
              type="checkbox"
              checked={task.completed}
              onchange={() => toggleTask(task)}
              class="w-5 h-5 rounded border-slate-300 text-indigo-500 focus:ring-indigo-400 cursor-pointer"
            />
            <span
              class="flex-1 text-slate-700 {task.completed ? 'line-through text-slate-400' : ''}"
            >
              {task.text}
            </span>
            <button
              on:click={() => deleteTask(task.id)}
              class="text-slate-400 hover:text-red-500 opacity-0 group-hover:opacity-100 transition-all duration-200 p-1 rounded hover:bg-red-50"
            >
              <svg
                xmlns="http://www.w3.org/2000/svg"
                class="h-5 w-5"
                fill="none"
                viewBox="0 0 24 24"
                stroke="currentColor"
                stroke-width="2"
              >
                <path
                  stroke-linecap="round"
                  stroke-linejoin="round"
                  d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16"
                />
              </svg>
            </button>
          </li>
        {/each}

        {#if filteredTasks.length === 0}
          <li class="px-4 py-10 text-center text-slate-400">
            {#if tasks.length === 0}
              <p class="text-lg">No tasks yet. Add one above!</p>
            {:else}
              <p class="text-lg">No {filter} tasks.</p>
            {/if}
          </li>
        {/if}
      </ul>

      <!-- Footer -->
      {#if tasks.length > 0}
        <div class="px-4 py-3 bg-slate-50 border-t border-slate-100 flex flex-col sm:flex-row items-center justify-between gap-3 text-sm text-slate-500">
          <span class="font-medium">{remaining} {remaining === 1 ? 'item' : 'items'} left</span>

          <div class="flex items-center gap-1">
            {#each ['all', 'active', 'completed'] as f}
              <button
                on:click={() => filter = f as 'all' | 'active' | 'completed'}
                class="px-3 py-1 rounded-md transition-colors duration-200 {filter === f ? 'bg-indigo-500 text-white font-medium' : 'hover:bg-slate-200 text-slate-600'}"
              >
                {f.charAt(0).toUpperCase() + f.slice(1)}
              </button>
            {/each}
          </div>

          {#if tasks.some((t) => t.completed)}
            <button
              on:click={clearCompleted}
              class="text-slate-400 hover:text-red-500 transition-colors duration-200 underline"
            >
              Clear completed
            </button>
          {/if}
        </div>
      {/if}
    </div>
  </div>
</div>
```

This single-file Svelte 5 + TypeScript todo app includes:

- **`$state` runes** for reactive state (`tasks`, `newTask`, `filter`)
- **`$derived`** for computed values (`filteredTasks`, `remaining`)
- **Add tasks** via the input field (press Enter or click "Add")
- **Toggle completion** with checkboxes
- **Delete** individual tasks (hover-reveal trash icon)
- **Filter** between All / Active / Completed
- **Remaining count** displayed in the footer
- **Clear completed** button (appears when there are completed tasks)
- **localStorage persistence** — tasks survive page reloads
- **Tailwind CSS** styling with a clean, modern UI (gradient background, card layout, smooth transitions, hover effects)

Drop this into a `.svelte` file in any Svelte 5 project with Tailwind configured.
