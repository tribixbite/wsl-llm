Here is the TypeScript implementation of the Sudoku solver using CSP (Constraint Satisfaction Problem) with AC-3 and Backtracking (MRV).

### TypeScript Implementation

```typescript
type Grid = number[][];
type Domain = number[][]; // 2D array where domain[r][c] is the list of possible numbers

// 1. Helper to generate the 81 variables (coordinates)
function getVariables(): [number, number][] {
  const vars: [number, number][] = [];
  for (let r = 0; r < 9; r++) {
    for (let c = 0; c < 9; c++) {
      vars.push([r, c]);
    }
  }
  return vars;
}

// 2. Helper to find neighbors (constraints) for a specific cell (r, c)
function getNeighbors(r: number, c: number): [number, number][] {
  const neighbors: [number, number][] = [];
  // Row
  for (let i = 0; i < 9; i++) if (i !== c) neighbors.push([r, i]);
  // Column
  for (let i = 0; i < 9; i++) if (i !== r) neighbors.push([i, c]);
  // 3x3 Box
  const boxStartRow = Math.floor(r / 3) * 3;
  const boxStartCol = Math.floor(c / 3) * 3;
  for (let i = boxStartRow; i < boxStartRow + 3; i++) {
    for (let j = boxStartCol; j < boxStartCol + 3; j++) {
      if ((i !== r || j !== c) && !neighbors.some(([nr, nc]) => nr === i && nc === j)) {
        neighbors.push([i, j]);
      }
    }
  }
  return neighbors;
}

// 3. AC-3 Algorithm
// Returns false if a domain becomes empty (inconsistency), true if stable
function ac3(grid: Grid, domains: Domain, queue: [number, number, number, number][]): boolean {
  while (queue.length > 0) {
    const [r1, c1, r2, c2] = queue.pop()!;
    
    // If we successfully remove a value from domain of X2 (r2, c2)
    if (remove(domains, r2, c2, grid[r1][c1])) {
      if (domains[r2][c2].length === 0) return false; // Domain wipeout

      // If X2 had only 1 value left, it's effectively assigned. 
      // We must enforce arc consistency from X2 to all its neighbors (except X1)
      if (domains[r2][c2].length === 1) {
        const assignedVal = domains[r2][c2][0];
        // Update grid temporarily to propagate constraints further if needed
        // (Though strictly AC-3 just maintains domains, updating grid helps logic flow)
        grid[r2][c2] = assignedVal; 

        const neighbors = getNeighbors(r2, c2);
        for (const [nr, nc] of neighbors) {
          // Don't add the arc back to the one we just processed
          if (nr === r1 && nc === c1) continue; 
          queue.push([r2, c2, nr, nc]);
        }
      }
    }
  }
  return true;
}

// Helper to remove a value from a domain
function remove(domains: Domain, r: number, c: number, value: number): boolean {
  const idx = domains[r][c].indexOf(value);
  if (idx !== -1) {
    domains[r][c].splice(idx, 1);
    return true;
  }
  return false;
}

// 4. Backtracking Search with MRV
let stepCount = 0;

function solve(grid: Grid, domains: Domain): boolean {
  // Check if solved (no 0s left)
  for (let r = 0; r < 9; r++) {
    for (let c = 0; c < 9; c++) {
      if (grid[r][c] === 0) return false;
    }
  }
  return true;

  // Select Variable: MRV Heuristic
  // Find the empty cell (0) with the smallest domain size
  let minDomainSize = 10;
  let selectedR = -1, selectedC = -1;

  for (let r = 0; r < 9; r++) {
    for (let c = 0; c < 9; c++) {
      if (grid[r][c] === 0) {
        if (domains[r][c].length < minDomainSize) {
          minDomainSize = domains[r][c].length;
          selectedR = r;
          selectedC = c;
        }
      }
    }
  }

  if (selectedR === -1) return true; // Should have been caught by solve check above, but safe fallback

  // Order Domains: Forward Checking / AC-3 integration
  // We iterate through the possible values for the selected variable
  const possibleValues = [...domains[selectedR][selectedC]];
  
  for (const val of possibleValues) {
    stepCount++;
    
    // Assign
    grid[selectedR][selectedC] = val;
    domains[selectedR][selectedC] = [val]; // Domain becomes singleton

    // Check consistency and propagate (AC-3)
    // Create a queue of arcs: (SelectedCell) -> (Neighbors)
    const queue: [number, number, number, number][] = [];
    const neighbors = getNeighbors(selectedR, selectedC);
    for (const [nr, nc] of neighbors) {
      queue.push([selectedR, selectedC, nr, nc]);
    }

    // Run AC-3
    // We pass a copy of domains to avoid mutating the original state if we backtrack,
    // OR we must deep copy domains. Deep copying 9x9 arrays is expensive.
    // Optimization: Since we modify grid and domains in place, we need to revert them on failure.
    
    // To implement this cleanly in JS/TS without heavy deep-copying overhead in the loop:
    // We will perform AC-3. If it fails, we revert.
    // Note: AC-3 modifies domains. We need to save the state of domains before AC-3 to revert?
    // Actually, simpler approach for Backtracking:
    // 1. Assign value.
    // 2. Run AC-3.
    // 3. If AC-3 returns false, backtrack (undo assignment).
    // 4. If AC-3 returns true, recurse.
    
    // To support reverting, we must copy domains before AC-3? 
    // Or, simpler: AC-3 is monotonic. If it fails, the state is ruined.
    // So we MUST deep copy domains before running AC-3 if we want to revert cleanly,
    // OR we implement a "reversible" AC-3 (complex).
    // Given the constraints, let's do a shallow copy of the domain arrays for the current step.
    
    const domainsSnapshot = domains.map(row => row.map(arr => [...arr]));

    const consistent = ac3(grid, domains, queue);

    if (consistent) {
      if (solve(grid, domains)) return true;
    }

    // Backtrack: Restore state
    grid[selectedR][selectedC] = 0;
    // Restore domains
    for(let r=0; r<9; r++) {
        for(let c=0; c<9; c++) {
            domains[r][c] = domainsSnapshot[r][c];
        }
    }
  }

  return false;
}

// --- Main Execution ---

const initialGrid: Grid = [
  [5,3,0,0,7,0,0,0,0],
  [6,0,0,1,9,5,0,0,0],
  [0,9,8,0,0,0,0,6,0],
  [8,0,0,0,6,0,0,0,3],
  [4,0,0,8,0,3,0,0,1],
  [7,0,0,0,2,0,0,0,6],
  [0,6,0,0,0,0,2,8,0],
  [0,0,0,4,1,9,0,0,5],
  [0,0,0,0,8,0,0,7,9]
];

// Deep copy grid to avoid modifying input
const grid = initialGrid.map(row => [...row]);

// Initialize Domains
const domains: Domain = [];
for (let r = 0; r < 9; r++) {
  const row: number[][] = [];
  for (let c = 0; c < 9; c++) {
    if (grid[r][c] !== 0) {
      row.push([grid[r][c]]); // Fixed value
    } else {
      row.push([1, 2, 3, 4, 5, 6, 7, 8, 9]); // All possibilities
    }
  }
  domains.push(row);
}

// Initial AC-3 pass to prune domains based on initial clues
const initialQueue: [number, number, number, number][] = [];
for (let r = 0; r < 9; r++) {
  for (let c = 0; c < 9; c++) {
    if (grid[r][c] !== 0) {
      const neighbors = getNeighbors(r, c);
      for (const [nr, nc] of neighbors) {
        initialQueue.push([r, c, nr, nc]);
      }
    }
  }
}

if (!ac3(grid, domains, initialQueue)) {
  console.log("Puzzle is unsolvable initially.");
} else {
  stepCount = 0;
  const solved = solve(grid, domains);
  
  if (solved) {
    console.log("Solved Board:");
    for (let r = 0; r < 9; r++) {
      console.log(grid[r].join(' '));
    }
    console.log(`Total Steps (Assignments attempted): ${stepCount}`);
  } else {
    console.log("No solution found.");
  }
}
```

### Explanation of the Algorithm

1.  **Representation**:
    *   **Variables**: Each cell $(r, c)$ in the 9x9 grid.
    *   **Domains**: Initially 1-9 for empty cells, or the specific number for filled cells.
    *   **Constraints**: All cells in the same row, column, or 3x3 box must have distinct values.

2.  **AC-3 (Arc Consistency)**:
    *   Before backtracking, we run AC-3 on the initial board. This removes values from the domains of empty cells that conflict with the pre-filled numbers.
    *   During backtracking, after assigning a value to a cell, we run AC-3 again. This prunes the domains of neighboring cells. If any domain becomes empty, the assignment is invalid, and we backtrack immediately. This is much more powerful than simple forward checking.

3.  **Backtracking with MRV**:
    *   **MRV (Minimum Remaining Values)**: Instead of picking the first empty cell, we pick the empty cell with the smallest domain size. This forces us to deal with the most constrained variables first, leading to faster failure detection if the path is wrong.
    *   **Step Count**: We increment `stepCount` every time we attempt to assign a value to a variable.

4.  **State Management**:
    *   Since AC-3 modifies the domains array in place, we must create a snapshot (deep copy) of the domains before running AC-3 during the recursive step. If the recursive call fails, we restore the domains from the snapshot to backtrack correctly.
